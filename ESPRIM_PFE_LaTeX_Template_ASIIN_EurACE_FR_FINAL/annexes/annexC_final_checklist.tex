\chapter{Checklist finale avant depot}

\section{Structure et coh\'erence}
\begin{itemize}[leftmargin=1.2cm]
\item \okitem{} Les pages liminaires sont compl\`etes.
\item \okitem{} L'introduction annonce la probl\'ematique, les objectifs et le plan.
\item \okitem{} Chaque chapitre a une logique interne claire.
\item \okitem{} La conclusion r\'epond explicitement aux objectifs.
\end{itemize}

\section{Qualite scientifique}
\begin{itemize}[leftmargin=1.2cm]
\item \okitem{} Les choix sont justifi\'es et compar\'es.
\item \okitem{} Les r\'esultats sont valid\'es par un protocole explicite.
\item \okitem{} Les limites et menaces sur la validit\'e sont discut\'ees.
\item \okitem{} Le rapport contient au moins \MinReferences{} r\'ef\'erences, avec une majorit\'e de sources scientifiques index\'ees.
\end{itemize}

\section{Forme et integrite}
\begin{itemize}[leftmargin=1.2cm]
\item \okitem{} Les figures et tableaux sont num\'erot\'es et cit\'es dans le texte.
\item \okitem{} Les unit\'es SI sont respect\'ees.
\item \okitem{} La bibliographie suit un style homog\`ene (IEEE recommand\'e).
\item \okitem{} Le taux de similarit\'e vis\'e reste \SimilarityTarget{}.
\item \okitem{} L'usage de l'IA g\'en\'erative est d\'eclar\'e si applicable.
\end{itemize}

\section{Fichiers a remettre}
\begin{itemize}[leftmargin=1.2cm]
\item PDF final du rapport.
\item Sources LaTeX (si demand\'ees par la fili\`ere).
\item Rapport de similarit\'e.
\item Support de soutenance.
\item Annexes techniques ou livrables compl\'ementaires si exig\'es.
\end{itemize}
